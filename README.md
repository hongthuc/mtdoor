    // ==================== SLIDER 2 BANNER TỰ TRƯỢT + HỖ TRỢ VUỐT CẢM ỨNG ====================
    let currentSlide = 0;
    const totalSlides = 2;
    let slideTimer = null;

    function updateSliderUI() {
      const track = document.getElementById('heroTrack');
      if (!track) return;
      track.style.transform = 'translateX(-' + (currentSlide * 100) + '%)';

      for (let i = 0; i < totalSlides; i++) {
        const dot = document.getElementById('dot-' + i);
        if (!dot) continue;
        if (i === currentSlide) {
          dot.className = 'w-6 h-2 rounded-full bg-white transition-all';
        } else {
          dot.className = 'w-2.5 h-2 rounded-full bg-white/45 transition-all';
        }
      }
    }

    function moveSlide(step) {
      currentSlide = (currentSlide + step + totalSlides) % totalSlides;
      updateSliderUI();
      restartAutoSlide();
    }

    function goToSlide(index) {
      currentSlide = index;
      updateSliderUI();
      restartAutoSlide();
    }

    function restartAutoSlide() {
      if (slideTimer) clearInterval(slideTimer);
      slideTimer = setInterval(() => {
        currentSlide = (currentSlide + 1) % totalSlides;
        updateSliderUI();
      }, 5000);
    }

    // Hỗ trợ vuốt trái/phải banner trên điện thoại
    function initSliderSwipe() {
      const slider = document.getElementById('heroSlider');
      if (!slider) return;
      let startX = 0;
      slider.addEventListener('touchstart', (e) => {
        startX = e.touches[0].clientX;
      }, { passive: true });
      slider.addEventListener('touchend', (e) => {
        const endX = e.changedTouches[0].clientX;
        const diff = startX - endX;
        if (Math.abs(diff) > 45) {
          moveSlide(diff > 0 ? 1 : -1);
        }
      }, { passive: true });
    }

    // ==================== LỌC SẢN PHẨM & TÌM KIẾM ====================
    function filterProductsByKeyword(keyword, btnEl) {
      const tabs = document.querySelectorAll('.filter-tab');
      tabs.forEach(t => {
        t.classList.remove('bg-brand-primary', 'text-white');
        t.classList.add('bg-white', 'text-gray-700', 'border', 'border-gray-200');
      });

      if (btnEl) {
        btnEl.classList.remove('bg-white', 'text-gray-700', 'border', 'border-gray-200');
        btnEl.classList.add('bg-brand-primary', 'text-white');
      } else if (keyword === 'all' && tabs[0]) {
        tabs[0].classList.remove('bg-white', 'text-gray-700', 'border', 'border-gray-200');
        tabs[0].classList.add('bg-brand-primary', 'text-white');
      }

      if (!keyword || keyword === 'all') {
        renderProducts(PRODUCTS);
        return;
      }

      const lower = keyword.toLowerCase().trim();
      const filtered = PRODUCTS.filter(p =>
        p.name.toLowerCase().includes(lower) ||
        p.tags.toLowerCase().includes(lower) ||
        p.specs.toLowerCase().includes(lower)
      );

      renderProducts(filtered);
    }
